import { SITE } from "../lib/site";
import { InstagramIcon, XIcon } from "./Icons";

type Props = {
  /** 表示の種類。header=アイコンのみ、inline=ラベル付き */
  variant?: "icons" | "inline";
  className?: string;
};

/**
 * Instagram・Xへのフォロー導線。
 * ヘッダー・ヒーロー下・毎日の問題・フッターで共通利用する。
 * 今後、問題ページ・記事ページにも同じコンポーネントを置く。
 */
export default function SnsFollow({ variant = "inline", className = "" }: Props) {
  if (variant === "icons") {
    return (
      <div className={`flex items-center gap-1 ${className}`}>
        <a
          href={SITE.instagramUrl}
          target="_blank"
          rel="noopener noreferrer"
          aria-label="Instagramを開く（新しいタブ）"
          className="grid h-11 w-11 place-items-center rounded-full text-white hover:text-samurai-gold transition"
        >
          <InstagramIcon className="h-6 w-6" />
        </a>
        <a
          href={SITE.xUrl}
          target="_blank"
          rel="noopener noreferrer"
          aria-label="Xを開く（新しいタブ）"
          className="grid h-11 w-11 place-items-center rounded-full text-white hover:text-samurai-gold transition"
        >
          <XIcon className="h-5 w-5" />
        </a>
      </div>
    );
  }

  return (
    <div className={`flex flex-wrap items-center gap-3 ${className}`}>
      <a
        href={SITE.instagramUrl}
        target="_blank"
        rel="noopener noreferrer"
        className="inline-flex min-h-11 items-center gap-2 rounded-full border-2 border-samurai-navy bg-white px-5 font-bold text-samurai-navy hover:bg-samurai-light transition"
      >
        <InstagramIcon className="h-5 w-5" />
        Instagramをフォロー
      </a>
      <a
        href={SITE.xUrl}
        target="_blank"
        rel="noopener noreferrer"
        className="inline-flex min-h-11 items-center gap-2 rounded-full border-2 border-samurai-navy bg-white px-5 font-bold text-samurai-navy hover:bg-samurai-light transition"
      >
        <XIcon className="h-4 w-4" />
        Xをフォロー
      </a>
    </div>
  );
}
